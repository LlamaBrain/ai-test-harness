cmake_minimum_required(VERSION 3.24)
project(ath_visual VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenCV 4.8 REQUIRED COMPONENTS core imgproc imgcodecs dnn)
find_package(nlohmann_json 3.11 REQUIRED)
add_library(ath_visual_lib src/analysis.cpp)
target_include_directories(ath_visual_lib PUBLIC src ${OpenCV_INCLUDE_DIRS})
target_link_libraries(ath_visual_lib PUBLIC ${OpenCV_LIBS} nlohmann_json::nlohmann_json)
add_executable(ath-visual src/main.cpp)
target_link_libraries(ath-visual PRIVATE ath_visual_lib)
include(CTest)
if(BUILD_TESTING)
  add_executable(ath_visual_digest_tests tests/batch_digest_test.cpp)
  add_test(NAME batch_result_digest COMMAND ath_visual_digest_tests)
  add_executable(ath_visual_tests tests/measure_test.cpp)
  target_link_libraries(ath_visual_tests PRIVATE ath_visual_lib)
  add_test(NAME pixel_measurement COMMAND ath_visual_tests)
  set_tests_properties(pixel_measurement PROPERTIES FIXTURES_SETUP golden_images)
  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_FOUND)
    add_test(NAME executable_integrity COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli_test.py $<TARGET_FILE:ath-visual> ${CMAKE_CURRENT_BINARY_DIR}/fixture-1080.png)
    set_tests_properties(executable_integrity PROPERTIES FIXTURES_REQUIRED golden_images)
  endif()
  add_executable(ath_visual_benchmark tests/benchmark.cpp)
  target_link_libraries(ath_visual_benchmark PRIVATE ath_visual_lib)
  if(WIN32)
    target_link_libraries(ath_visual_benchmark PRIVATE psapi)
  endif()
endif()
